def reinforce_related(memories: List[EventMemory]) -> None:
    """Похожие события усиливают друг друга (resonance)."""
    for i, m1 in enumerate(memories):
        for m2 in memories[i+1:]:
            if m1.target_id == m2.target_id and m1.emotion_tag == m2.emotion_tag:
                # Резонанс: повторяющийся паттерн → higher importance
                m1.importance = min(1.0, m1.importance + 0.1)
                m2.importance = min(1.0, m2.importance + 0.1)

@dataclass
class NPCScheme:
    """Активная схема NPC против игрока или другого NPC."""
    scheme_type: str        # "murder", "blackmail", "befriend", "seduce"
    target_id: str
    progress: float         # 0.0–1.0
    secrecy: float          # 0.0–1.0 (если < 0.3, "замечено")
    collaborators: Set[str]  # npc_ids в сговоре
    
    def on_event(self, event: EventMemory) -> None:
        """Событие может продвинуть или сломать схему."""
        if event.target_id == self.target_id and event.event_type == "betrayal":
            self.progress += 0.2  # цель предала кого-то — легче манипулировать

class ResonanceEngine:
    """
    Анализирует паттерны в EventMemory NPC.
    Создаёт InternalVoice, WorldLegend, NPCScheme автоматически.
    """
    def analyze(self, npc: NPCState) -> ResonanceResult:
        memories = get_all_memories(npc)
        
        # Паттерн: 3+ предательства от разных людей → "paranoia" voice
        betrayals = [m for m in memories if m.event_type == "betrayal"]
        if len(betrayals) >= 3:
            return AddInternalVoice("paranoia", "Все предают. Не доверяй никому.")
        
        # Паттерн: помогал игроку 5+ раз → "loyalty" voice + схема защиты
        helps = [m for m in memories if m.target_id == "player" and m.event_type == "help"]
        if len(helps) >= 5:
            return AddScheme("protect_player", target="player", secrecy=0.9)
        
        # Паттерн: событие затронуло 10+ NPC → WorldLegend
        if self.count_participants(memories[-1]) > 10:
            return CreateWorldLegend(memories[-1])
