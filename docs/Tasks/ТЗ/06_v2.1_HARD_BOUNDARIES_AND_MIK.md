### 1. Жёсткие инварианты (Global Invariants)

```python
class InvariantKernel:
    """
    Что в системе НИКОГДА не может быть нарушено?
    Эти инварианты — закон, не предложение.
    """
    
    INVARIANT_1_SINGLE_CAUSAL_TICK = True
    """
    Каузальный тик ВСЕГДА инкрементируется на 1.
    Никогда не замедляется, никогда не ускоряется.
    Нарушение = crash immediately
    """
    
    INVARIANT_2_STATE_OWNERSHIP = {
        "position": "SpatialService",           # Единственный владелец
        "perception": "PerceptualKernel",       # Единственный владелец
        "decision": "DecisionHub",              # Единственный владелец
        "emotion": "AffectiveIntegrator",       # Единственный владелец
        # Каждый владелец имеет монополию на изменение
    }
    """
    Каждое состояние имеет одного владельца.
    Никакая другая система не может его менять напрямую.
    Нарушение = architecture error
    """
    
    INVARIANT_3_PRIORITY_RULE = {
        "conflict_between_times": "causal > biological > narrative",
        "conflict_between_domains": "ownership > temporal > perception",
        "conflict_between_values": "integrity > consistency > plausibility",
    }
    """
    Когда два источника истины расходятся —
    это правило определяет, кто побеждает.
    Нарушение = state divergence
    """
    
    INVARIANT_4_CAUSAL_CLOSURE = True
    """
    Всё состояние должно быть объяснимо через цепь причин.
    Нет "магических" состояний, появившихся откуда-то.
    Нарушение = ghost state
    """
    
    INVARIANT_5_COUPLING_HARD_LIMIT = 0.4
    """
    Никакой уровень не может иметь coupling score > 0.4
    Это не метрика, это constraint.
    Нарушение = immediate refactoring trigger
    """
```

### 2. Minimal Invariant Kernel (MIK)

```python
class MinimalInvariantKernel:
    """
    Самое маленькое ядро, которое гарантирует консистентность.
    Всё остальное — над этим ядром.
    """
    
    def __init__(self):
        # Три базовых объекта, от которых зависит ВСЁ
        self.causal_state: CausalSimulationCore
        self.ownership_registry: StateOwnershipRegistry
        self.invariant_checker: InvariantValidator
    
    def apply_change(self, change: StateChange) -> bool:
        """
        Любое изменение состояния проходит через MIK.
        Если оно нарушает инварианты → ОТКАЗАТЬ.
        """
        # 1. Проверить: кто пытается менять?
        owner = self.ownership_registry.get_owner(change.attribute)
        if change.source != owner:
            raise ArchitectureError(
                f"Only {owner} can change {change.attribute}, "
                f"not {change.source}"
            )
        
        # 2. Проверить: не нарушает ли инварианты?
        if not self.invariant_checker.validate(change):
            raise InvariantViolation(
                f"Change violates: {change.violated_invariants}"
            )
        
        # 3. Проверить: causal tick правильный?
        if change.tick != self.causal_state.current_tick:
            raise CausalityError(
                f"Change must be applied at tick {self.causal_state.current_tick}, "
                f"not {change.tick}"
            )
        
        # 4. Применить изменение
        self.causal_state.apply_change(change)
        return True
```

### 3. Временная схема (простая, не философская)

```python
class TimeScheduling:
    """
    Не 'три типа времени', а 'три layer update frequency'
    """
    
    REACTIVE_LAYER_UPDATE = 1        # Каждый тик
    SITUATIONAL_LAYER_UPDATE = 10    # Каждые 10 тиков
    ADAPTIVE_LAYER_UPDATE = 100      # Каждые 100 тиков
    STRUCTURAL_LAYER_UPDATE = 1000   # Каждые 1000 тиков
    
    def update_all_layers(self, current_tick: int):
        """Вместо 'temporal domain' — просто частота обновления"""
        if current_tick % self.REACTIVE_LAYER_UPDATE == 0:
            self._update_reactive_layer()
        
        if current_tick % self.SITUATIONAL_LAYER_UPDATE == 0:
            self._update_situational_layer()
        
        if current_tick % self.ADAPTIVE_LAYER_UPDATE == 0:
            self._update_adaptive_layer()
        
        if current_tick % self.STRUCTURAL_LAYER_UPDATE == 0:
            self._update_structural_layer()
    
    # Это всё. Нет философии, есть исполняемый график.
```

### 4. Coupling Enforcement (не метрика, а constraint)

```python
class CouplingEnforcement:
    """
    Coupling score не просто наблюдается — он ограничен архитектурой.
    """
    
    MAX_DEPENDENCIES_PER_ENTITY = 3
    
    def register_dependency(self, entity: str, depends_on: str) -> bool:
        """
        При попытке добавить зависимость:
        1. Проверить текущий score
        2. Если score > MAX → отказать попытку
        3. Заставить переделать архитектуру
        """
        current_deps = self.get_current_dependencies(entity)
        if len(current_deps) >= self.MAX_DEPENDENCIES_PER_ENTITY:
            raise ArchitectureConstraintError(
                f"{entity} уже имеет {len(current_deps)} зависимостей. "
                f"Новая: {depends_on}. "
                f"Нужен рефакторинг."
            )
        
        self.dependencies[entity].add(depends_on)
        return True
```

### 5. Causal Grounding (не психология, а физика)

```python
class CausalGrounding:
    """
    Все состояния должны быть объяснены через цепь причин.
    """
    
    def validate_state_causality(self, npc: NPC):
        """
        Каждое поле состояния должно иметь объяснение:
        "Почему fear = 0.8?"
        Ответ должен быть цепью причин, прослеживаемой до события.
        """
        
        # Пример валидного объяснения:
        causality = {
            "fear": {
                "source": "combat_event",
                "tick": 145,
                "magnitude": 0.5,
                "decay_since": 10,  # 10 тиков назад
                "current_value": 0.5 * (1 - 0.1 * 10) = 0.3,  # Восстановилась
            }
        }
        
        # Если fear = 0.8, но:
        # - Нет event
        # - Нет decay
        # - Нет объяснения
        # → INVALID STATE
        
        # Система должна отказать такое состояние
        # или выявить баг, создавший phantom state
```
