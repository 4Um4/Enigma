  183: def transition_commitment(
  184:     commitment: Dict[str, Any],
  185:     new_status: str,
  186:     tick: Optional[int] = None,
  187:     interrupt_reason: Optional[str] = None,
  188:     fail_reason: Optional[str] = None,
  189: ) -> bool:
  190:     """FSM-переход обязательства. True = разрешён и выполнен.
  191: 
  192:     INTERRUPTED без interrupt_reason запрещён: прерывание обязано нести
  193:     причину прекращения. S203.4 (D-6): FAILED без fail_reason запрещён —
  194:     симметрия; провал обязан нести причину (cause ≠ interrupt_reason
  195:     ≠ fail_reason, №7).
  196:     """
  197:     current: str = commitment.get("status", "")
  198:     allowed = COMMITMENT_TRANSITIONS.get(current, set())
  199:     if new_status not in allowed:
  200:         return False
  201:     if new_status == "INTERRUPTED" and not interrupt_reason:
  202:         return False
  203:     if new_status == "FAILED" and not fail_reason:
  204:         return False
  205: 
  206:     commitment["status"] = new_status
  207:     if new_status == "INTERRUPTED":
  208:         commitment["interrupt_reason"] = interrupt_reason
  209:     if new_status == "FAILED":
  210:         commitment["fail_reason"] = fail_reason
  211:     # S203.4 (Ц5): часы BLOCKED-фазы для таймаут-продюсера (Э7). Вход в
  212:     # BLOCKED заводит часы (tick=None → часы не заведены: sweep такой записи
  213:     # не таймаутит — честное отсутствие сигнала, не ноль); любой выход из
  214:     # BLOCKED (EXECUTING/терминалы) — сброс.
  215:     if new_status == "BLOCKED":
  216:         commitment["blocked_since_tick"] = tick
  217:     elif current == "BLOCKED":
  218:         commitment["blocked_since_tick"] = None
  219:     if tick is not None:
  220:         commitment["updated_tick"] = tick
  221:     return True
